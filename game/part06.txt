}
function ownedSettlements(w,id){return w.settlements.filter(s=>s.owner===id);}
function neighborsOf(w,id){const ids=[];for(const [a,b] of w.roads){if(a===id)ids.push(b);else if(b===id)ids.push(a);}return ids.map(x=>w.settlements[x]);}
function adminPolicyFor(policy='balanced'){
 if(policy==='economy')return {build:['farms','lumber','mine','keep','hall','barracks','walls'],units:['militia','spears']};
 if(policy==='defense')return {build:['walls','keep','barracks','farms','lumber','mine','hall'],units:['spears','militia','cavalry']};
 if(policy==='military')return {build:['barracks','farms','mine','keep','walls','lumber','hall'],units:['raiders','spears','cavalry','militia']};
 return {build:['farms','lumber','mine','barracks','walls','hall','keep'],units:['spears','militia','raiders','cavalry']};
}

 // Human-scale administration: a realm gets a small capped number of decisions, not one per town.
 if(w.day-(f.lastAdminDay??-99)>=5){
   const slots=Math.min(3,1+Math.floor(Math.max(0,owns.length-1)/8));
   const ordered=owns.slice().sort((a,b)=>(a.buildings.farms+a.buildings.lumber+a.buildings.mine+a.buildings.keep)-(b.buildings.farms+b.buildings.lumber+b.buildings.mine+b.buildings.keep));
   for(let n=0;n<slots;n++){const s=ordered[(f.adminCursor+n)%ordered.length];if(s)runAdminSlot(w,id,s,rng,'balanced');}
   f.adminCursor=(f.adminCursor+slots)%Math.max(1,ordered.length);f.lastAdminDay=w.day;
 }
 // Diplomacy every ~20 days. Sovereign wars cannot begin during the opening truce.
 if(w.day>=FOUNDING_TRUCE_DAYS&&w.day%20===id%5){for(const other of w.factions){if(other.id===id||!other.alive)continue;let rel=getRelation(w,id,other.id),status=getDiplomacy(w,id,other.id);const my=factionPower(w,id),op=factionPower(w,other.id);rel+=rng()<0.5?-2:2;if(status==='peace'&&rel>58&&rng()<0.2){setDiplomacy(w,id,other.id,'alliance',rel);w.events.push({day:w.day,type:'diplomacy',actors:[id,other.id],text:`${f.name} and ${other.name} formed an alliance.`});}
 else if(status==='war'&&rel>-20&&rng()<0.3){setDiplomacy(w,id,other.id,'peace',rel+10);w.events.push({day:w.day,type:'diplomacy',actors:[id,other.id],text:`${f.name} and ${other.name} agreed to peace.`});}
 else if(status==='peace'&&rel<-45&&my>op*0.8&&rng()<0.25){setDiplomacy(w,id,other.id,'war',rel-10);w.events.push({day:w.day,type:'diplomacy',actors:[id,other.id],text:`${f.name} declared war on ${other.name}.`});}
 else setDiplomacy(w,id,other.id,status,rel);}}
 // One strategic dispatch per command window, with a cap on simultaneous offensives.
 if(w.day-(f.lastMilitaryDay??-99)<8)return;
 const active=w.armies.filter(a=>a.owner===id).length,maxActive=Math.min(3,1+Math.floor(owns.length/7));if(active>=maxActive)return;

 if(!candidates.length)return;candidates.sort((a,b)=>b.ratio-a.ratio||a.score-b.score);const pick=candidates[Math.floor(rng()*Math.min(3,candidates.length))];const src=pick.from,tgt=pick.to;for(const [u,q] of (src.buildings.barracks>=2?[['raiders',4],['spears',4],['militia',5]]:[['spears',4],['militia',5]])){if(recruit(w,src.id,u,q).ok)break;}const available=armyPower(src.troops,true);const need=settlementDefense(tgt);if(available<need*0.74){f.lastMilitaryDay=w.day;return;}
 const frac=available>need*1.7?0.64:0.78,units={};for(const k of Object.keys(UNITS))units[k]=Math.floor(src.troops[k]*frac);const hostile=tgt.owner!==null;const relation=hostile?getRelation(w,id,tgt.owner):0;if(hostile&&getDiplomacy(w,id,tgt.owner)!=='war'&&relation>-40&&rng()<0.6){f.lastMilitaryDay=w.day;return;}

}
