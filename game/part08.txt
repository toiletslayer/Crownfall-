function eventScope(e){
  const actors=Array.isArray(e.actors)?e.actors:[];
  if(actors.includes(0))return 'player';
  if(actors.some(id=>id!==null&&id!==0&&getDiplomacy(world,0,id)==='alliance'))return 'ally';
  const text=String(e.text||'');
  if(text.includes(world.factions[0].name)||/\byour\b|\bcrown is yours\b/i.test(text))return 'player';
  for(const f of world.factions.slice(1))if(getDiplomacy(world,0,f.id)==='alliance'&&text.includes(f.name))return 'ally';
  return 'world';
}
