}
  else{const relevant=[...world.events].reverse().find(e=>eventScope(e)!=='world'&&world.day-e.day<=6);if(relevant){const scope=eventScope(relevant);notice.className=`mapNotice ${scope}`;notice.innerHTML=`<span>${scope==='player'?'YOUR REALM':'ALLY'}</span><b>Day ${relevant.day}</b> ${eventIcon(relevant.type)} ${relevant.text}`;notice.classList.remove('hidden');}else notice.classList.add('hidden');}

let lastMapStructureSig='';
function mapStructureSig(){
  return String(world.seed)+'|'+world.settlements.length+'|'+world.roads.length;
}
function syncMapNotice(threats){
  const notice=$('#mapNotice'); if(!notice)return;
  if(threats.length){
    const a=threats[0],target=world.settlements[a.target];
    notice.className='mapNotice threat';
    notice.innerHTML=`<span>⚠ INCOMING</span><b>${target.name}</b> ${world.factions[a.owner].name} ${a.mission} force · ${armySize(a.units)} troops · ETA ${Math.max(0,a.arriveDay-world.day)} days`;
    notice.classList.remove('hidden');
    return;
  }
  const relevant=[...world.events].reverse().find(e=>eventScope(e)!=='world'&&world.day-e.day<=6);
  if(relevant){
    const scope=eventScope(relevant);
    notice.className=`mapNotice ${scope}`;
    notice.innerHTML=`<span>${scope==='player'?'YOUR REALM':'ALLY'}</span><b>Day ${relevant.day}</b> ${eventIcon(relevant.type)} ${relevant.text}`;
    notice.classList.remove('hidden');
  }else notice.className='mapNotice hidden';
}
function updateMapTick(){
  const map=$('#map'); if(!map)return;
  const sig=mapStructureSig();
  if(!map.querySelector('.settlement')||sig!==lastMapStructureSig){renderMap();return;}
  const threats=incomingThreats(),threatened=new Set(threats.map(a=>a.target));
  map.querySelectorAll('.settlement').forEach(el=>{
    const id=Number(el.dataset.id),s=world.settlements[id]; if(!s)return;
    let stance='neutral';
    if(s.owner===0)stance='playerRealm';
    else if(s.owner!==null){const d=getDiplomacy(world,0,s.owner);stance=d==='alliance'?'alliedRealm':d==='war'?'enemyRealm':'foreignRealm';}
    ['neutral','playerRealm','alliedRealm','enemyRealm','foreignRealm'].forEach(k=>el.classList.toggle(k,k===stance));
    el.classList.toggle('capital',!!s.capital);
    el.classList.toggle('selected',id===selected);
    el.classList.toggle('incomingThreat',threatened.has(id));
    el.title=`${s.name} — ${ownerName(s)}\nDefense ${Math.floor(settlementDefense(s))}`;
  });
  map.querySelectorAll('.label').forEach(el=>{
    const prev=el.previousElementSibling;
    const id=prev?.classList?.contains('settlement')?Number(prev.dataset.id):NaN,s=world.settlements[id];
    if(!s)return;
    el.classList.toggle('label-player',s.owner===0);
    el.classList.toggle('label-neutral',s.owner===null);
    el.classList.toggle('label-foreign',s.owner!==null&&s.owner!==0);
    el.classList.toggle('label-selected',id===selected);
  });
  const roadEls=$('#roads .road');
  roadEls.forEach((el,i)=>{const edge=world.roads[i];if(!edge)return;const A=world.settlements[edge[0]],B=world.settlements[edge[1]];el.classList.toggle('road-player',A.owner===0||B.owner===0);});
  $('#legend').innerHTML=world.factions.map(f=>{const lands=world.settlements.filter(s=>s.owner===f.id).length;return `<span class="legendItem ${lands?'':'eliminated'}"><i class="dot" style="background:${f.color}"></i>${f.name} <small>${lands?lands+' lands':'fallen'}</small></span>`;}).join('')+'<span class="legendItem"><i class="dot" style="background:#aaa58d"></i>Independent</span>';
  map.querySelectorAll('.army').forEach(el=>el.remove());
  for(const a of world.armies){
    const A=world.settlements[a.from],B=world.settlements[a.target];
    const total=Math.max(1,a.arriveDay-a.departDay),p=Math.max(0,Math.min(1,(world.day-a.departDay)/total));
    const x=A.x+(B.x-A.x)*p,y=A.y+(B.y-A.y)*p;
    const stance=a.owner===0?'playerArmy':getDiplomacy(world,0,a.owner)==='alliance'?'alliedArmy':getDiplomacy(world,0,a.owner)==='war'?'enemyArmy':'foreignArmy';
    const el=document.createElement('div');el.className=`army ${stance}`;el.style.left=x+'%';el.style.top=y+'%';el.style.background=world.factions[a.owner].color;
    el.title=`${world.factions[a.owner].name} — ${a.mission} — ${armySize(a.units)} troops — ETA ${Math.max(0,a.arriveDay-world.day)}d`;map.appendChild(el);
  }
  syncMapNotice(threats);
}

function nearestOwnedOrigin(targetId){
  let best=null;for(const s of owned()){const p=shortestPath(world,s.id,targetId);if(!p)continue;const score=p.length;if(!best||score<best.score)best={s,score};}return best?.s||owned()[0]||null;
}
function prepareOrder(targetId,mission){
  const origin=nearestOwnedOrigin(targetId);if(!origin)return status('You have no settlement from which to issue orders.');
  const d=getArmyDraft(origin);d.target=targetId;d.mission=mission;selected=origin.id;activeTab='settlement';renderAll();status(`${mission[0].toUpperCase()+mission.slice(1)} prepared from ${origin.name} against ${world.settlements[targetId].name}. Choose troop numbers and dispatch.`);
  $('#armyTarget')?.scrollIntoView({block:'nearest'});
}
function intelFor(s){
  const paths=owned().map(o=>shortestPath(world,o.id,s.id)).filter(Boolean).map(p=>p.length-1);const hops=paths.length?Math.min(...paths):99;
  if(hops<=1)return 'detailed';if(hops<=3)return 'partial';return 'distant';
}
